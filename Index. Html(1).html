        };

        const greetingTextEl = document.getElementById('greetingText');

        const totalCountEl = document.getElementById('totalCount');

        // Setup Tanggal & Waktu
        function setupDateAndGreeting() {
            const now = new Date();
            const options = { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' };
            currentDateEl.textContent = now.toLocaleDateString('id-ID', options);

            const hour = now.getHours();
            let greeting = "Selamat malam,";
            if (hour >= 5 && hour < 11) greeting = "Selamat pagi,";
            else if (hour >= 11 && hour < 15) greeting = "Selamat siang,";
            else if (hour >= 15 && hour < 18) greeting = "Selamat sore,";

            greetingTextEl.textContent = `${greeting} semoga harimu menyenangkan!`;
        }

        // Format waktu (09:00)
        function formatTime(timeStr) {
            return timeStr; // Input type time sudah mengembalikan format HH:MM
        }

        // Simpan ke LocalStorage
        function saveTasks() {
            localStorage.setItem('pudyy_schedules', JSON.stringify(tasks));
            renderTasks();
        }

        // Tambah Jadwal Baru
        taskForm.addEventListener('submit', function(e) {
            e.preventDefault();
           
            const name = document.getElementById('taskName').value;
            const time = document.getElementById('taskTime').value;
            const category = document.querySelector('input[name="category"]:checked').value;

            const newTask = {
                id: Date.now().toString(),
                name: name,
                time: time,
                category: category,
                completed: false
            };

            tasks.push(newTask);
           
            // Urutkan berdasarkan waktu
            tasks.sort((a, b) => a.time.localeCompare(b.time));
           
            saveTasks();
            taskForm.reset();
            // Kembalikan radio button ke default
            document.querySelector('input[name="category"][value="rutinitas"]').checked = true;
        });

        // Hapus Jadwal
        function deleteTask(id) {
            // Animasi hilang
            const el = document.getElementById(`task-${id}`);
            el.style.opacity = '0';
            el.style.transform = 'translateX(20px)';
           
            setTimeout(() => {
                tasks = tasks.filter(task => task.id !== id);
                saveTasks();
            }, 300); // Sesuaikan dengan durasi transisi
        }

        // Toggle Status Selesai
        function toggleTask(id) {
            const task = tasks.find(t => t.id === id);
            if (task) {
                task.completed = !task.completed;
                saveTasks();
            }
        }
